  <script>
    const systemSelect = document.getElementById('systemSelect');
    const weeksInput = document.getElementById('weeks');
    const weeklyPayment = document.getElementById('weeklyPayment');

    const systemPrices = { basic: 900, standard: 1500, family: 2500 };
    const initialDeposits = { basic: 225, standard: 375, family: 625 };

    function calculatePayment() {
      const selectedSystem = systemSelect.value;
      const totalWeeks = parseInt(weeksInput.value, 10);

      if (totalWeeks > 0) {
        const totalCost = systemPrices[selectedSystem];
        const deposit = initialDeposits[selectedSystem];
        const remainingBalance = totalCost - deposit;
        const weeklyCost = remainingBalance / totalWeeks;
        weeklyPayment.textContent = `GH₵ ${weeklyCost.toFixed(2)}`;
      } else {
        weeklyPayment.textContent = 'GH₵ 0.00';
      }
    }

    systemSelect.addEventListener('change', calculatePayment);
    weeksInput.addEventListener('input', calculatePayment);
    document.addEventListener('DOMContentLoaded', calculatePayment);
  </script>
